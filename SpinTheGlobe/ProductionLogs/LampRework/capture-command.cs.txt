using System; using System.IO; using System.Linq; using System.Collections.Generic; using UnityEngine; using UnityEditor; using UnityEngine.SceneManagement; using UnityEditor.SceneManagement; using DetectiveDesk.Editor;
internal class CommandScript:IRunCommand {
void Render(Camera cam,string path,int w,int h) {
var saved=EditorJsonUtility.ToJson(cam); var target=cam.targetTexture; var active=RenderTexture.active; float aspect=cam.aspect;
var rt=new RenderTexture(w,h,24,RenderTextureFormat.ARGB32);Texture2D image=null;
try {rt.Create();cam.targetTexture=rt;cam.aspect=(float)w/h;cam.rect=new Rect(0,0,1,1);cam.Render();RenderTexture.active=rt;image=new Texture2D(w,h,TextureFormat.RGB24,false);image.ReadPixels(new Rect(0,0,w,h),0,0);image.Apply();File.WriteAllBytes(path,image.EncodeToPNG());}
finally {cam.targetTexture=target;cam.aspect=aspect;EditorJsonUtility.FromJsonOverwrite(saved,cam);RenderTexture.active=active;if(image)UnityEngine.Object.DestroyImmediate(image);rt.Release();UnityEngine.Object.DestroyImmediate(rt);}
}
public void Execute(ExecutionResult result) {
var scene=SceneManager.GetActiveScene();var camera=GameObject.Find("DetectiveOffice/MainCamera").GetComponent<Camera>();
var canvases=scene.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<Canvas>(true)).ToArray();
var rects=scene.GetRootGameObjects().SelectMany(x=>x.GetComponentsInChildren<RectTransform>(true)).ToArray();
var states=canvases.ToDictionary(x=>x,x=>EditorJsonUtility.ToJson(x));var rs=rects.ToDictionary(x=>x,x=>EditorJsonUtility.ToJson(x));
var cameraState=EditorJsonUtility.ToJson(camera);
try {foreach(var c in canvases.Where(x=>x.renderMode==RenderMode.ScreenSpaceOverlay)){c.renderMode=RenderMode.ScreenSpaceCamera;c.worldCamera=camera;c.planeDistance=.5f;} Canvas.ForceUpdateCanvases();Render(camera,"Assets/DetectiveDesk/LampRework/final-game-camera.png",1920,1080);}
finally {foreach(var p in states)EditorJsonUtility.FromJsonOverwrite(p.Value,p.Key);foreach(var p in rs)EditorJsonUtility.FromJsonOverwrite(p.Value,p.Key);EditorJsonUtility.FromJsonOverwrite(cameraState,camera);}
Scene review=default(Scene);GameObject go=null;
try {review=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);SceneManager.SetActiveScene(review);go=new GameObject("LampRework_TemporaryCloseupCamera");var close=go.AddComponent<Camera>();close.CopyFrom(camera);close.targetTexture=null;close.fieldOfView=44;close.transform.position=new Vector3(4.8f,1.87f,-1.8f);close.transform.LookAt(new Vector3(4.25f,1.60f,.53f));
var originalData=camera.GetComponent<UnityEngine.Rendering.Universal.UniversalAdditionalCameraData>();if(originalData){var data=go.AddComponent<UnityEngine.Rendering.Universal.UniversalAdditionalCameraData>();EditorUtility.CopySerialized(originalData,data);}
Render(close,"Assets/DetectiveDesk/LampRework/unity-lamp-closeup.png",1080,1080);}
finally {SceneManager.SetActiveScene(scene);if(review.IsValid())EditorSceneManager.CloseScene(review,true);}
LampReworkIntegration.Preservation();result.Log("Game camera with HUD and additive camera closeup captured. All original serialized fields restored and preservation passed.");
}}
