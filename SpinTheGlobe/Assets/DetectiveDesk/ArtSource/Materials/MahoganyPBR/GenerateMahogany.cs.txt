using System;
using System.IO;
using UnityEngine;
using UnityEditor;
internal class CommandScript:IRunCommand {
const string Folder="Assets/DetectiveDesk/AssetProduction/MahoganyPBR/20261030_215053_f81465af";
const int N=1024;
float Height(float u,float v) {
float warp=.012f*Mathf.Sin(2*Mathf.PI*v)+.005f*Mathf.Sin(6*Mathf.PI*v);
float x=u+warp;
float grain=Mathf.Sin(2*Mathf.PI*(61*x+ .15f*Mathf.Sin(4*Mathf.PI*v)));
float fine=Mathf.Sin(2*Mathf.PI*(173*x+ .3f*Mathf.Sin(6*Mathf.PI*v)));
return .5f+.22f*grain+.09f*fine+.1f*Mathf.Sin(2*Mathf.PI*(9*x+ .25f*Mathf.Sin(2*Mathf.PI*v)));
}
void Save(string name,Color[] pixels) {var t=new Texture2D(N,N,TextureFormat.RGBA32,false,true);t.SetPixels(pixels);t.Apply();File.WriteAllBytes(Folder+"/"+name+".png",t.EncodeToPNG());UnityEngine.Object.DestroyImmediate(t);}
Texture2D Import(string name,bool srgb,bool normal) {
string p=Folder+"/"+name+".png";AssetDatabase.ImportAsset(p,ImportAssetOptions.ForceSynchronousImport);
var i=(TextureImporter)AssetImporter.GetAtPath(p);i.textureType=normal?TextureImporterType.NormalMap:TextureImporterType.Default;i.sRGBTexture=srgb;i.wrapMode=TextureWrapMode.Repeat;i.mipmapEnabled=true;i.maxTextureSize=N;i.textureCompression=TextureImporterCompression.Uncompressed;i.SaveAndReimport();return AssetDatabase.LoadAssetAtPath<Texture2D>(p);
}
public void Execute(ExecutionResult result) {
Directory.CreateDirectory(Folder);
if(File.Exists(Folder+"/MahoganyPBR.mat"))throw new Exception("Refusing overwrite");
var shader=Shader.Find("Universal Render Pipeline/Lit");if(shader==null||!shader.isSupported)throw new Exception("URP Lit unavailable");
var color=new Color[N*N];var rough=new Color[N*N];var normal=new Color[N*N];var packed=new Color[N*N];
float minR=1,maxR=0;
for(int y=0;y<N;y++)for(int x=0;x<N;x++) {
float u=(float)x/N,v=(float)y/N,h=Height(u,v);
float ribbon=.5f+.5f*Mathf.Sin(2*Mathf.PI*(5*u+.18f*Mathf.Sin(2*Mathf.PI*v)));
float pore=Mathf.Pow(Mathf.Max(0,Mathf.Sin(2*Mathf.PI*(227*u+.5f*Mathf.Sin(8*Mathf.PI*v)))),18)*(.5f+.5f*Mathf.Sin(2*Mathf.PI*37*v));
float tone=.64f+.26f*h+.10f*ribbon-.11f*pore;
int k=y*N+x;color[k]=new Color(.27f*tone,.102f*tone,.052f*tone,1);
float r=Mathf.Clamp(.34f+.14f*(1-h)+.08f*pore,.3f,.6f);minR=Mathf.Min(minR,r);maxR=Mathf.Max(maxR,r);
rough[k]=new Color(r,r,r,1);packed[k]=new Color(0,0,0,1-r);
float dx=(Height(u+1f/N,v)-Height(u-1f/N,v))*.12f;
float dy=(Height(u,v+1f/N)-Height(u,v-1f/N))*.12f;
var n=new Vector3(-dx,-dy,1).normalized;normal[k]=new Color(n.x*.5f+.5f,n.y*.5f+.5f,n.z*.5f+.5f,1);
}
Save("Mahogany_BaseColor",color);Save("Mahogany_Roughness",rough);Save("Mahogany_Normal",normal);Save("Mahogany_MetallicSmoothness",packed);
var b=Import("Mahogany_BaseColor",true,false);var rtex=Import("Mahogany_Roughness",false,false);var ntex=Import("Mahogany_Normal",false,true);var mtex=Import("Mahogany_MetallicSmoothness",false,false);
var m=new Material(shader);m.name="MahoganyPBR";m.SetTexture("_BaseMap",b);m.SetColor("_BaseColor",Color.white);m.SetTexture("_BumpMap",ntex);m.SetFloat("_BumpScale",1);m.EnableKeyword("_NORMALMAP");m.SetTexture("_MetallicGlossMap",mtex);m.SetFloat("_Metallic",0);m.SetFloat("_Smoothness",1);m.SetFloat("_SmoothnessTextureChannel",0);m.EnableKeyword("_METALLICSPECGLOSSMAP");AssetDatabase.CreateAsset(m,Folder+"/MahoganyPBR.mat");
AssetDatabase.SaveAssetIfDirty(m);
File.WriteAllText(Folder+"/review_manifest.json","{\"kind\":\"material\",\"asset_paths\":[\""+Folder+"/MahoganyPBR.mat\",\""+Folder+"/Mahogany_BaseColor.png\",\""+Folder+"/Mahogany_Roughness.png\",\""+Folder+"/Mahogany_Normal.png\",\""+Folder+"/Mahogany_MetallicSmoothness.png\"],\"prefab_path\":\""+Folder+"/MahoganyPBR_Review.prefab\"}");
File.WriteAllText(Folder+"/material-validation.json","{\"status\":\"PASS\",\"resolution\":1024,\"tileable_analytic_source\":true,\"roughness_min\":"+minR.ToString(System.Globalization.CultureInfo.InvariantCulture)+",\"roughness_max\":"+maxR.ToString(System.Globalization.CultureInfo.InvariantCulture)+",\"metallic\":0,\"normal_import\":true,\"shader\":\"Universal Render Pipeline/Lit\"}");
result.Log("Generated and imported four 1024 maps and standalone URP material; roughness "+minR+" to "+maxR);
}
}
