using System.IO;
using UnityEngine;
using UnityEngine.UI;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
internal class CommandScript:IRunCommand {
const string Folder="Assets/DetectiveDesk/Art/UI/InteractionFeedback/20261030_214853_b1cdac91";
static Color Brass=new Color(.68f,.51f,.25f,1), Green=new Color(.23f,.57f,.39f,1);
static GameObject Node(string name,Transform parent,Vector2 size,Vector2 pos) {
var go=new GameObject(name,typeof(RectTransform)); go.transform.SetParent(parent,false);
var r=go.GetComponent<RectTransform>();r.sizeDelta=size;r.anchoredPosition=pos;return go;
}
static Text Label(string text,Transform parent,Vector2 size,Vector2 pos,int fontSize,Color color) {
var go=Node(text,parent,size,pos);var t=go.AddComponent<Text>();t.text=text;t.font=Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");t.fontSize=fontSize;t.alignment=TextAnchor.MiddleCenter;t.color=color;t.raycastTarget=false;return t;
}
static GameObject Control(Transform parent,Vector2 pos,Sprite sprite,Color glow,string label) {
var root=Node("InteractionFeedbackControl",parent,new Vector2(320,86),pos);
var halo=Node("SoftFeedback",root.transform,new Vector2(344,110),Vector2.zero).AddComponent<UnityEngine.UI.Image>();
halo.sprite=sprite;halo.type=UnityEngine.UI.Image.Type.Sliced;halo.color=glow;halo.raycastTarget=false;
var plate=Node("Plate",root.transform,new Vector2(320,86),Vector2.zero).AddComponent<UnityEngine.UI.Image>();plate.color=new Color(.065f,.052f,.045f,1);
var edge=plate.gameObject.AddComponent<Outline>();edge.effectColor=new Color(.4f,.32f,.2f,.65f);edge.effectDistance=new Vector2(1,-1);
Label(label,root.transform,new Vector2(300,62),Vector2.zero,20,new Color(.85f,.81f,.7f));
var button=root.AddComponent<Button>();button.targetGraphic=halo;
var c=button.colors;c.normalColor=new Color(.48f,.38f,.24f,.12f);c.highlightedColor=Brass;c.selectedColor=Green;c.pressedColor=new Color(.35f,.72f,.48f,1);c.disabledColor=new Color(.25f,.25f,.25f,.06f);c.colorMultiplier=1;c.fadeDuration=.14f;button.colors=c;
return root;
}
public void Execute(ExecutionResult result) {
var original=SceneManager.GetActiveScene();var temp=default(Scene);
try {
temp=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);SceneManager.SetActiveScene(temp);
var tex=new Texture2D(64,64,TextureFormat.RGBA32,false);tex.name="SoftRectHalo";
for(int y=0;y<64;y++)for(int x=0;x<64;x++){float d=Mathf.Min(x,y,63-x,63-y);float a=Mathf.SmoothStep(0,.42f,d/12f);tex.SetPixel(x,y,new Color(1,1,1,a));}
tex.Apply();File.WriteAllBytes(Folder+"/SoftRectHalo.png",tex.EncodeToPNG());Object.DestroyImmediate(tex);
AssetDatabase.ImportAsset(Folder+"/SoftRectHalo.png",ImportAssetOptions.ForceSynchronousImport);
var imp=(TextureImporter)AssetImporter.GetAtPath(Folder+"/SoftRectHalo.png");imp.textureType=TextureImporterType.Sprite;imp.spriteImportMode=SpriteImportMode.Single;imp.spriteBorder=new Vector4(16,16,16,16);imp.alphaIsTransparency=true;imp.mipmapEnabled=false;imp.wrapMode=TextureWrapMode.Clamp;imp.SaveAndReimport();
var sprite=AssetDatabase.LoadAssetAtPath<Sprite>(Folder+"/SoftRectHalo.png");
var canvasGo=new GameObject("InteractionFeedbackStateSheet",typeof(RectTransform),typeof(Canvas),typeof(CanvasScaler));
var canvas=canvasGo.GetComponent<Canvas>();canvas.renderMode=RenderMode.ScreenSpaceOverlay;
var scaler=canvasGo.GetComponent<CanvasScaler>();scaler.uiScaleMode=CanvasScaler.ScaleMode.ScaleWithScreenSize;scaler.referenceResolution=new Vector2(1200,900);
Label("INTERACTION / FEEDBACK",canvasGo.transform,new Vector2(900,70),new Vector2(0,290),30,new Color(.78f,.65f,.4f));
Label("AGED BRASS   /   BANKER GREEN",canvasGo.transform,new Vector2(900,50),new Vector2(0,230),16,new Color(.6f,.61f,.57f));
string[] names={"REST","HOVER","SELECTED"};Color[] colors={new Color(.48f,.38f,.24f,.12f),Brass,Green};
for(int i=0;i<3;i++){
float y=110-i*150;
var control=Control(canvasGo.transform,new Vector2(0,y),sprite,colors[i],"INSPECT EVIDENCE");
if(i==0){control.GetComponent<RectTransform>().anchoredPosition=Vector2.zero;control.transform.Find("SoftFeedback").GetComponent<UnityEngine.UI.Image>().color=Color.white;PrefabUtility.SaveAsPrefabAsset(control,Folder+"/InteractionFeedbackControl.prefab");control.GetComponent<RectTransform>().anchoredPosition=new Vector2(0,y);}
Object.DestroyImmediate(control.GetComponent<Button>());control.transform.Find("SoftFeedback").GetComponent<UnityEngine.UI.Image>().color=colors[i];
Label(names[i],canvasGo.transform,new Vector2(230,50),new Vector2(-330,y),16,new Color(.65f,.65f,.59f));
}
Label("Hover to discover. Select to hold focus.",canvasGo.transform,new Vector2(850,60),new Vector2(0,-350),19,new Color(.64f,.65f,.6f));
PrefabUtility.SaveAsPrefabAsset(canvasGo,Folder+"/InteractionFeedbackStateSheet.prefab");
File.WriteAllText(Folder+"/review_manifest.json","{\"kind\":\"ui\",\"asset_paths\":[\""+Folder+"/InteractionFeedbackStateSheet.prefab\",\""+Folder+"/InteractionFeedbackControl.prefab\",\""+Folder+"/SoftRectHalo.png\"],\"prefab_path\":\""+Folder+"/InteractionFeedbackReview.prefab\"}");
var asset=AssetDatabase.LoadAssetAtPath<GameObject>(Folder+"/InteractionFeedbackControl.prefab");
var b=asset.GetComponent<Button>();if(!b||!b.targetGraphic||b.colors.selectedColor!=Green||b.colors.highlightedColor!=Brass)throw new System.Exception("Control configuration invalid");
File.WriteAllText(Folder+"/ui-authoring-validation.json","{\"status\":\"PASS\",\"button_color_states_checked\":true,\"texture_dimensions\":[64,64],\"control_dimensions_ui_units\":[320,86],\"pivot\":[0.5,0.5],\"play_mode_tested\":false}");
result.Log("Standalone UI prefab, texture and state sheet authored; serialized Button states checked.");
} finally {if(original.IsValid())SceneManager.SetActiveScene(original);if(temp.IsValid()&&temp.isLoaded)EditorSceneManager.CloseScene(temp,true);}
}
}
