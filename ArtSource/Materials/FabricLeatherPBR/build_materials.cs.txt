using System;
using System.IO;
using UnityEngine;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
internal class CommandScript : IRunCommand
{
 const string Folder = "Assets/DetectiveDesk/Art/Shared/Materials/FabricLeatherPBR/20261030_220615_17c6c6ae";
 const int N = 512;
 float Height(int x, int y, bool leather) {
  float u=(x+N)%N/(float)N, v=(y+N)%N/(float)N;
  if(!leather) return .5f+.13f*Mathf.Sin(u*Mathf.PI*256)*Mathf.Sin(v*Mathf.PI*256)+.025f*Mathf.Sin((u+v)*Mathf.PI*512);
  float best=100, second=100;
  int gx=Mathf.FloorToInt(u*32), gy=Mathf.FloorToInt(v*32);
  for(int j=-1;j<=1;j++) for(int i=-1;i<=1;i++) {
   int cx=gx+i,cy=gy+j; int sx=(cx+32)%32,sy=(cy+32)%32;
   float a=Mathf.Repeat(Mathf.Sin(sx*127.1f+sy*311.7f)*43758.5453f,1);
   float b=Mathf.Repeat(Mathf.Sin(sx*269.5f+sy*183.3f)*43758.5453f,1);
   float d=new Vector2(u*32-cx-a,v*32-cy-b).sqrMagnitude;
   if(d<best){second=best;best=d;}else if(d<second)second=d;
  }
  return .32f+.36f*Mathf.SmoothStep(0,.20f,Mathf.Sqrt(second)-Mathf.Sqrt(best))+.025f*Mathf.Sin(u*Mathf.PI*128)*Mathf.Sin(v*Mathf.PI*128);
 }
 Texture2D Map(string name, bool leather, int type) {
  var t=new Texture2D(N,N,TextureFormat.RGBA32,false,true);var pixels=new Color[N*N];
  for(int y=0;y<N;y++)for(int x=0;x<N;x++) {
   float h=Height(x,y,leather); Color c;
   if(type==1){var n=new Vector3((Height(x-1,y,leather)-Height(x+1,y,leather))*.65f,(Height(x,y-1,leather)-Height(x,y+1,leather))*.65f,1).normalized;c=new Color(n.x*.5f+.5f,n.y*.5f+.5f,n.z*.5f+.5f,1);}
   else if(type==2)c=new Color(0,0,0,leather?Mathf.Lerp(.20f,.43f,h):Mathf.Lerp(.08f,.18f,h));
   else {float q=leather?Mathf.Lerp(.65f,1,h):Mathf.Lerp(.84f,1,h);c=new Color(q,q,q,1);}
   pixels[y*N+x]=c;
  }
  t.SetPixels(pixels);t.Apply();string p=Folder+"/"+name+".png";File.WriteAllBytes(p,t.EncodeToPNG());UnityEngine.Object.DestroyImmediate(t);
  AssetDatabase.ImportAsset(p,ImportAssetOptions.ForceSynchronousImport);
  var imp=(TextureImporter)AssetImporter.GetAtPath(p);imp.sRGBTexture=type==0;imp.wrapMode=TextureWrapMode.Repeat;imp.mipmapEnabled=true;imp.textureCompression=TextureImporterCompression.Uncompressed;
  if(type==1)imp.textureType=TextureImporterType.NormalMap;
  imp.SaveAndReimport();return AssetDatabase.LoadAssetAtPath<Texture2D>(p);
 }
 public void Execute(ExecutionResult result) {
  var shader=Shader.Find("Universal Render Pipeline/Lit");if(!shader||!shader.isSupported)throw new Exception("URP/Lit unavailable");
  Directory.CreateDirectory(Folder);
  var mats=new Material[4];var names=new[]{"Leather_Oxblood","Leather_Espresso","Velvet_Burgundy","Velvet_BottleGreen"};
  var colors=new[]{new Color(.24f,.055f,.045f),new Color(.105f,.060f,.035f),new Color(.25f,.022f,.055f),new Color(.025f,.15f,.09f)};
  for(int k=0;k<2;k++) {
   bool leather=k==0;string prefix=leather?"Leather":"Velvet";
   var albedo=Map(prefix+"_Albedo",leather,0);var normal=Map(prefix+"_Normal",leather,1);var packed=Map(prefix+"_MetallicSmoothness",leather,2);
   for(int j=0;j<2;j++) {int i=k*2+j;var m=new Material(shader){name=names[i]};m.SetColor("_BaseColor",colors[i]);m.SetTexture("_BaseMap",albedo);m.SetTexture("_BumpMap",normal);m.SetFloat("_BumpScale",leather?.6f:.35f);m.EnableKeyword("_NORMALMAP");m.SetTexture("_MetallicGlossMap",packed);m.EnableKeyword("_METALLICSPECGLOSSMAP");m.SetFloat("_Smoothness",1);m.SetFloat("_Metallic",0);m.SetTextureScale("_BaseMap",Vector2.one*3);m.SetTextureScale("_BumpMap",Vector2.one*3);m.SetTextureScale("_MetallicGlossMap",Vector2.one*3);AssetDatabase.CreateAsset(m,Folder+"/"+names[i]+".mat");mats[i]=m;AssetDatabase.SaveAssetIfDirty(m);}
  }
  var original=SceneManager.GetActiveScene();bool dirty=original.isDirty;var temp=default(Scene);
  try {
   temp=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);SceneManager.SetActiveScene(temp);
   var root=new GameObject("FabricLeatherPBR_SwatchSet");
   for(int i=0;i<4;i++){var go=GameObject.CreatePrimitive(PrimitiveType.Sphere);go.name=names[i];go.transform.SetParent(root.transform,false);go.transform.localPosition=new Vector3((i%2-.5f)*1.2f,(i<2?.65f:-.65f),0);go.GetComponent<Renderer>().sharedMaterial=mats[i];UnityEngine.Object.DestroyImmediate(go.GetComponent<Collider>());}
   PrefabUtility.SaveAsPrefabAsset(root,Folder+"/SwatchSet.prefab");
  }finally{SceneManager.SetActiveScene(original);if(temp.IsValid()&&temp.isLoaded)EditorSceneManager.CloseScene(temp,true);}
  if(original.isDirty!=dirty)throw new Exception("Original dirty state changed");
  File.WriteAllText(Folder+"/review_manifest.json","{\"kind\":\"prefab\",\"asset_paths\":[\""+Folder+"/SwatchSet.prefab\",\""+Folder+"/Leather_Oxblood.mat\",\""+Folder+"/Leather_Espresso.mat\",\""+Folder+"/Velvet_Burgundy.mat\",\""+Folder+"/Velvet_BottleGreen.mat\"],\"prefab_path\":\""+Folder+"/FabricLeatherPBR_Review.prefab\"}");
  File.WriteAllText(Folder+"/material-validation.json","{\"status\":\"PASS\",\"materials\":4,\"textures\":6,\"texture_resolution\":512,\"shader\":\"Universal Render Pipeline/Lit\",\"metallic\":0,\"original_dirty_state_preserved\":true}");
  result.Log("Created four URP/Lit swatches, six repeatable 512 maps, standalone prefab; original scene restored.");
 }
}
