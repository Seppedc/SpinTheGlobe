using System;
using System.IO;
using UnityEngine;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
internal class CommandScript : IRunCommand {
 public void Execute(ExecutionResult result) {
 string folder="Assets/DetectiveDesk/Art/Props/CluePhotos/20261031_014924_21ace326";
 string[] names={"RainyPassage","WarehouseWindows","UnknownVisitor"};
 var original=SceneManager.GetActiveScene(); bool dirty=original.isDirty;
 var temporary=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);
 try {
 SceneManager.SetActiveScene(temporary);
 var root=new GameObject("CluePhotos_Display");
 var shader=Shader.Find("Universal Render Pipeline/Lit");
 if(!shader || !shader.isSupported) throw new Exception("URP Lit unavailable");
 for(int n=0;n<3;n++){
 int w=768,h=1024;var tex=new Texture2D(w,h,TextureFormat.RGB24,false);
 var pixels=new Color32[w*h];var rng=new System.Random(2600+n);
 for(int y=0;y<h;y++)for(int x=0;x<w;x++){
 float u=(float)x/w,v=(float)y/h;
 float grain=(float)rng.NextDouble()-.5f;
 float value=.78f+grain*.055f;
 bool image=u>.065f&&u<.935f&&v>.16f&&v<.945f;
 if(image){
 float a=(u-.065f)/.87f,b=(v-.16f)/.785f;
 value=.32f+.21f*b;
 if(n==0){
 float half=.055f+.48f*(1-b);
 if(Mathf.Abs(a-.5f)>half) {
 value=.105f+.05f*Mathf.PerlinNoise(a*48,b*27);
 if((Mathf.FloorToInt(a*19)%4==1)&&(Mathf.FloorToInt(b*13)%3==1))value=.46f;
 } else if(b<.52f) value=.13f+.18f*Mathf.Abs(Mathf.Sin(b*180))* (1-b);
 if(Mathf.Abs(a-.53f)<.007f&&b>.35f&&b<.85f)value=.07f;
 if(Mathf.Pow((a-.53f)/.055f,2)+Mathf.Pow((b-.84f)/.025f,2)<1)value=.85f;
 }else if(n==1){
 value=.23f+Mathf.PerlinNoise(a*80,b*70)*.09f;
 float bx=Mathf.Repeat(a*4,.999f),by=Mathf.Repeat(b*3,.999f);
 if(b>.24f&&bx>.16f&&bx<.83f&&by>.2f&&by<.75f){
 value=.57f;
 if(Mathf.Abs(bx-.5f)<.025f||Mathf.Abs(by-.48f)<.025f)value=.10f;
 }
 if(b<.24f&&a>.37f&&a<.68f)value=.065f;
 if(Mathf.Repeat(b*40,1)<.06f)value*=.7f;
 }else {
 value=.62f;
 if(a<.14f||a>.86f||Mathf.Abs(a-.5f)<.015f||Mathf.Abs(b-.57f)<.015f)value=.12f;
 float head=Mathf.Pow((a-.52f)/.10f,2)+Mathf.Pow((b-.61f)/.12f,2);
 if(head<1||(b>.62f&&b<.66f&&a>.36f&&a<.68f)||(b<.51f&&Mathf.Abs(a-.52f)<(.51f-b)*.48f+.07f))value=.065f;
 if(b<.13f)value=.13f;
 }
 value+=grain*.11f;
 // Fine vertical rain and subdued silver-gelatin vignette.
 if((x*37+n*13)%113<2&&y%83<42)value+=.09f;
 value*=1-.35f*Mathf.Clamp01((a-.5f)*(a-.5f)*2+(b-.5f)*(b-.5f));
 }else {
 // Worn paper edge and three distinct pencil-like accession marks.
 if(u<.012f||u>.988f||v<.012f||v>.988f)value-=.12f;
 if(v>.075f&&v<.079f&&u>.12f&&u<.3f+n*.08f)value=.31f;
 }
 value=Mathf.Clamp01(value);
 pixels[y*w+x]=new Color(value*.98f,value*.93f,value*.82f,1);
 }
 tex.SetPixels32(pixels);tex.Apply();
 string path=folder+"/"+names[n]+".png";File.WriteAllBytes(path,tex.EncodeToPNG());UnityEngine.Object.DestroyImmediate(tex);
 AssetDatabase.ImportAsset(path,ImportAssetOptions.ForceSynchronousImport);
 var importer=(TextureImporter)AssetImporter.GetAtPath(path);importer.textureType=TextureImporterType.Default;importer.sRGBTexture=true;importer.mipmapEnabled=true;importer.wrapMode=TextureWrapMode.Clamp;importer.maxTextureSize=1024;importer.textureCompression=TextureImporterCompression.Uncompressed;importer.SaveAndReimport();
 var material=new Material(shader);material.name="Photo_"+names[n];material.SetTexture("_BaseMap",AssetDatabase.LoadAssetAtPath<Texture2D>(path));material.SetFloat("_Smoothness",.16f);material.SetFloat("_Metallic",0);material.SetFloat("_Cull",0);
 AssetDatabase.CreateAsset(material,folder+"/"+material.name+".mat");
 var quad=GameObject.CreatePrimitive(PrimitiveType.Quad);quad.name="Photo_"+names[n];quad.transform.SetParent(root.transform,false);quad.transform.localPosition=new Vector3((n-1)*.105f,0,0);quad.transform.localScale=new Vector3(.09f,.12f,1);quad.GetComponent<Renderer>().sharedMaterial=material;UnityEngine.Object.DestroyImmediate(quad.GetComponent<Collider>());
 PrefabUtility.SaveAsPrefabAsset(quad,folder+"/"+names[n]+".prefab");
 }
 PrefabUtility.SaveAsPrefabAsset(root,folder+"/CluePhotos_Display.prefab");
 File.WriteAllText(folder+"/review_manifest.json","{\"kind\":\"prefab\",\"asset_paths\":[\""+folder+"/CluePhotos_Display.prefab\",\""+folder+"/RainyPassage.png\",\""+folder+"/WarehouseWindows.png\",\""+folder+"/UnknownVisitor.png\"],\"prefab_path\":\""+folder+"/CluePhotos_Review.prefab\"}");
 File.WriteAllText(folder+"/authoring-validation.json","{\"status\":\"PASS\",\"texture_count\":3,\"texture_dimensions\":\"768x1024\",\"paper_dimensions_m\":[0.09,0.12],\"pivot\":\"center\",\"shader\":\"Universal Render Pipeline/Lit\",\"colliders\":0,\"runtime_scripts\":0}");
 } finally {SceneManager.SetActiveScene(original);EditorSceneManager.CloseScene(temporary,true);}
 if(original.isDirty!=dirty)throw new Exception("Original scene dirty state changed");
 result.Log("Authored three separate textures, materials and photo prefabs; restored original scene and dirty state.");
 }
}
